#include <tunables/global>

profile sharp_ac_gateway flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  signal (receive) peer=unconfined,
  signal (receive) peer=*_sharp_ac_gateway,
  signal (send) set=(kill,term,int),

  # Stream sockets are limited by code to Supervisor and MQTT.
  network unix stream,
  network inet stream,

  # IPv4 datagrams cover bounded DNS and the direct ECHONET UDP endpoint.
  network inet dgram,
  deny network inet6 dgram,
  deny network inet6 stream,
  deny network inet raw,
  deny network inet6 raw,
  deny network packet,
  deny network netlink raw,
  deny network netlink dgram,

  /run.sh rix,
  /bin/sh rix,
  /usr/bin/python3* rix,
  /usr/local/bin/python3* rix,
  /usr/local/lib/ r,
  /usr/local/lib/** rm,
  /opt/sharp-ac/ r,
  /opt/sharp-ac/** r,
  /opt/sharp-ac/sharp_ac/** rm,
  /data/ rw,
  /data/** rwk,
  /tmp/ rw,
  /tmp/** rwk,

  /dev/null rw,
  /dev/urandom r,
  /etc/group r,
  /etc/hosts r,
  /etc/nsswitch.conf r,
  /etc/passwd r,
  /etc/resolv.conf r,
  /etc/ssl/** r,
  /proc/*/fd/ r,
  /proc/sys/crypto/fips_enabled r,
}
